```javascript

    名前: 型(初期値, {オプション}),
    名前: {
        // ネストする時は{}を使う。
    },

```

```
const o = tob({
    name: tob.str,
    age: tob.int,
    is: tob.bln,
    class: tob.str('初期値'),
    items: tob.ary.int(1,2),
    value: tob.int({min:0, max:100}),
    nest: {
        group: {
            name: tob.str,
            obj: tob.pob.has('hasKey','hasKey2').notHas('持ってはならないキー', '...'),
        },
    },
});
o.name = 0; // 代入時点で即座に例外発生。
const errMsgs = o.veryfy(); // 特定APIでまとめて実行。複数のエラーメッセージを取得する。

}
```
